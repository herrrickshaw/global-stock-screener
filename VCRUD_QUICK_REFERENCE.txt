COMMON TASKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

┌─ Find Duplicates ─────────────────────────────────────────────────────────┐
│ git checkout feature/vcrud-system-tracker                                 │
│ python find_duplicates_standalone.py                                      │
│ # Output: Shows all duplicate files across branches                       │
└───────────────────────────────────────────────────────────────────────────┘

┌─ Compare Your Branch to Main ─────────────────────────────────────────────┐
│ git checkout feature/vcrud-system-tracker                                 │
│ python vcrud_cli.py compare --branch1 main --branch2 your-branch          │
│ # Output: Files only in main, only in your branch, common                 │
└───────────────────────────────────────────────────────────────────────────┘

┌─ Get File Statistics ─────────────────────────────────────────────────────┐
│ python vcrud_cli.py stats --branch your-branch                            │
│ # Output: File count, size, compression ratio                             │
└───────────────────────────────────────────────────────────────────────────┘

┌─ Optimize Storage ────────────────────────────────────────────────────────┐
│ python vcrud_cli.py optimize --branch your-branch                         │
│ # Output: Recommendations for storage optimization                        │
└───────────────────────────────────────────────────────────────────────────┘

┌─ List All Files on Branch ────────────────────────────────────────────────┐
│ python vcrud_cli.py list --branch your-branch                             │
│ # Output: All files with sizes (top 50)                                   │
└───────────────────────────────────────────────────────────────────────────┘

┌─ Export Analysis to JSON ─────────────────────────────────────────────────┐
│ python vcrud_cli.py export --branch your-branch -o report.json            │
│ # Output: Detailed report for further analysis                            │
└───────────────────────────────────────────────────────────────────────────┘

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Add to ~/.zshrc or ~/.bash_profile:

vcrud-check() {
  local branch=$(git rev-parse --abbrev-ref HEAD)
  git stash
  git checkout feature/vcrud-system-tracker
  python find_duplicates_standalone.py
  git checkout "$branch"
  git stash pop 2>/dev/null
}

vcrud-compare() {
  local branch=$(git rev-parse --abbrev-ref HEAD)
  git checkout feature/vcrud-system-tracker
  python vcrud_cli.py compare --branch1 main --branch2 "$branch"
  git checkout "$branch"
}

Then use:
  vcrud-check     # Find duplicates
  vcrud-compare   # Compare with main
